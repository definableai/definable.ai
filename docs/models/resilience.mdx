`ResilientModel` is a wrapper, not a model subclass. It delegates calls to an inner model, rotates API keys on rate limits, and can fail over to backup providers when the current model is unavailable.

## Tested behavior

```python
from dataclasses import dataclass, field
from typing import Any

from definable.exceptions import ModelProviderError, ModelRateLimitError
from definable.model.base import Model
from definable.model.resilience.failover import FailoverChain, FailoverEntry
from definable.model.resilience.key_pool import KeyPool
from definable.model.resilience.resilient import ResilientModel
from definable.model.response import ModelResponse


@dataclass
class FakeModel(Model):
    id: str = "primary"
    name: str = "FakeModel"
    provider: str = "fake"
    api_key: str = ""
    client: Any = None
    async_client: Any = None
    errors: list[Any] = field(default_factory=list)

    def invoke(self, *args: Any, **kwargs: Any) -> ModelResponse:
        if self.errors:
            error = self.errors.pop(0)
            if error is not None:
                raise error
        return ModelResponse(content="ok")

    async def ainvoke(self, *args: Any, **kwargs: Any) -> ModelResponse:
        return self.invoke(*args, **kwargs)

    def invoke_stream(self, *args: Any, **kwargs: Any):
        yield ModelResponse(content="chunk")

    async def ainvoke_stream(self, *args: Any, **kwargs: Any):
        yield ModelResponse(content="chunk")

    def _parse_provider_response(self, response: Any, **kwargs: Any) -> ModelResponse:
        return ModelResponse(content="parsed")

    def _parse_provider_response_delta(self, response: Any, **kwargs: Any) -> ModelResponse:
        return ModelResponse(content="delta")


rotations = []
failovers = []
primary = FakeModel(errors=[ModelRateLimitError("slow down"), ModelProviderError("down", status_code=500)])
backup = FakeModel(id="backup")

model = ResilientModel(
    inner=primary,
    key_pool=KeyPool(keys=["sk-1", "sk-2"], base_cooldown=0, max_cooldown=0),
    failover=FailoverChain([FailoverEntry(model=backup, priority=1)]),
    on_key_rotated=rotations.append,
    on_failover=failovers.append,
)

result = model.invoke()

assert result.content == "ok"
assert model.id == "backup"
assert len(rotations) == 1
assert len(failovers) == 1
assert primary.api_key == "sk-2"
```

## What each piece does

- `KeyPool` owns the available keys and their cooldown state.
- `ResilientModel` rotates keys only for rate-limit style failures.
- `FailoverChain` is an ordered list of backup models, sorted by priority.

## Notes

- `MCPClient.connect()` style best-effort behavior does not apply here. If every provider is exhausted, `ResilientModel` raises a `ModelProviderError`.
- The wrapper mirrors the active inner model's `id`, `name`, and `provider`, so telemetry reflects the provider that actually served the request.
