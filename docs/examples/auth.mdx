## Tested auth providers

```python
import asyncio

from definable.agent.auth import APIKeyAuth, AllowlistAuth, CompositeAuth
from definable.agent.auth.base import AuthRequest


async def main() -> None:
    api_keys = APIKeyAuth(keys={"secret-key"})
    allowlist = AllowlistAuth(user_ids={"telegram-user"}, platforms={"telegram"})
    composite = CompositeAuth(api_keys, allowlist)

    http_context = await composite.authenticate(
        AuthRequest(platform="http", headers={"X-API-Key": "secret-key"})
    )
    telegram_context = await composite.authenticate(
        AuthRequest(platform="telegram", user_id="telegram-user")
    )

    assert http_context is not None
    assert http_context.user_id.startswith("apikey_")
    assert telegram_context is not None
    assert telegram_context.user_id == "telegram-user"


asyncio.run(main())
```
